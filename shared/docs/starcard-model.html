GraphQL ที่ API ต้องมี (ร่าง · ยังไม่ได้ทำ)
    
# ข้อมูลเจ้าของที่ widget ทุกหมวดอ่าน — ให้ทีม API ดูว่าต้องเก็บอะไร
# ✅ = มีใน API แล้ว (อ่านจาก path ท้ายบรรทัด) · ❗ = ยังไม่มี ต้องเพิ่ม
# แอปแปลงเป็น field ของ Model เอง (24800 → "24.8K") ชื่อฝั่ง widget ไม่ผูกกับชื่อ API

extend type User { starOwner: StarOwner! }        # StarCard.owner: User → ไม่เก็บซ้ำในการ์ด

type StarOwner {
  # ฐาน — ทุกหมวด
  name: String!                 # ✅ myProfile.name
  handle: String                # ✅ socialProfiles[].profileUrl (ตัด username)
  isVerified: Boolean!          # ✅ myProfile.isVerify            → owner.verified · หมวด seal
  profileImages: [Image!]!      # ✅ creatorProfile.profileImages   → รูปช่อง 1–3
  workImages: [Image!]!         # ✅ creatorProfile.portfolio       → รูปช่อง 4+ · หมวด photo/showcase

  # hero · intro · tags
  nickname: String              # ✅ myProfile.nickname
  about: String                 # ✅ myProfile.caption
  # tagline ไม่ต้องมี — ไม่มีที่กรอก แอปสร้างจาก categories เอง ("แฟชั่น · คาเฟ่")
  workArea: String              # ✅ creatorProfile.preferredProvinces[0]
  creatorType: String           # ❗ creator | page (ฟิลด์ใหม่ตัวเดียวของ Star Profile) → stat:creatorKind “บุคคล/เพจ”
  categories: [String!]!        # ✅ categories(limit: 20).name

  socials: [StarSocial!]!       # followers · rate
  audience: StarAudience        # ❗ audience ทั้งหมวด
  trackRecord: StarTrackRecord! # brand · verified
  contact: StarContact!         # contact
  availability: StarAvailability!
  body: StarBody                # ✅ creatorProfile.bodyMeasurement
  kycStatus: String!            # ✅ userVerify.status             → ล็อก/ปลดตรา
}

type StarSocial {
  platform: SocialProfileType!  # ✅ type                         ❗ ยังไม่มี x
  profileUrl: String!           # ✅
  username: String              # ✅ accountId / displayName
  followerCount: Int!           # ✅ followerCount
  avgViewCount: Int             # ✅ (query เดิมไม่ได้ดึง)
  avgEngagementCount: Int       # ✅ (query เดิมไม่ได้ดึง)
  syncedAt: Date                # ❗ มีแค่ statusSyncFollower
  isLinked: Boolean!            # ✅ OAuth แล้ว vs กรอกเอง         → socials.verified
  photoPrice: Int               # ✅ รวม short/long/sharing         → rates[]
  shortVdoPrice: Int
  longVdoPrice: Int
  sharingPrice: Int
}

type StarAudience {             # ❗ ทั้ง type — ตอนนี้ MOCK
  female: Float!  male: Float!  other: Float!      # % เพศ
  ages:   [Share!]!             # { label: "18–24", share: 42 }
  places: [Share!]!             # { label: "กรุงเทพฯ", share: 61 }
  reach: StarReach              # { accounts delta newShare window platform }
}

type StarTrackRecord {          # ข้อเสนอเดิมใน star-card.schema.graphql
  completedCampaigns: Int!      # ❗
  brands: [StarBrand!]!         # ❗ { name logo industry campaigns }
  works: [StarWork!]!           # ❗ { brand campaign ep platform cover views saves shares likes comments }
}

type StarContact {
  tel: String      showTel: Boolean!        # ✅ creatorProfile
  lineId: String   showLineId: Boolean!     # ✅
  website: String  showWebsite: Boolean!    # ✅ (ยังไม่ส่งเข้า owner)
  email: String                             # ❗
}

type StarAvailability {
  days: [CreatorAvailableDays!]!            # ✅ availableTimes.day
  slots: [CreatorTimeSlots!]!               # ✅ availableTimes.timeSlots
  provinces: [String!]!                     # ✅ preferredProvinces
}

type StarBody {                 # ✅ ทั้ง type = bodyMeasurement {value unit}
  weight: Measure  height: Measure  chest: Measure  waist: Measure  hip: Measure  shoeSize: Measure
}

    วาดตอนไหนบ้าง
    
    
    แต่ละหน้าในแอปวาดการ์ด/widget ไม่เหมือนกัน — บางหน้าวาดสด บางหน้าแค่โชว์รูปที่อบไว้
    
    
    
      
    หน้าวาดยังไง
    
      
        
    หน้าแต่ง / ดูการ์ดวาดสด ทุกชิ้น ทุกหน้า
    
        
    ตู้ widget (+)วาดสด เฉพาะช่องที่เห็นบนจอ ด้วยข้อมูลจริงหรือข้อมูลตัวอย่าง (หัวข้อที่ยังไม่กรอก)
    
        
    Star Card ของฉัน · การ์ดใน Star Profileวาดสด แล้วย่อทั้งใบ
    
        
    เลือก Templateรูปที่อบไว้ ไม่วาด · แตะเลือกแล้วค่อยวาดสดในหน้าแต่ง
    
        
    แชร์ / ส่งออกวาดนอกจอ ที่ความละเอียดสูง แล้วบันทึกเป็น PNG
    
          
    
    

    
    ระหว่างแต่ง: ทำอะไรแล้ววาดใหม่ขั้นไหน
      
    
        
    ทำอะไรเกิดอะไรขึ้นขั้นที่ทำชิ้นไหน
    
        
          
    เปิดหน้า Star Cardโหลดการ์ด → วาดพื้นหลังผืนเดียว → วางทุกหน้า → วาดทุกชิ้น1 2 3 4ทุกชิ้น
    
          
    แตะเลือก widget ในตู้สร้างชิ้นใหม่ขนาดตั้งต้น → หาที่ว่าง (หน้านี้ → หน้าถัดไป → ย่อได้ถึง 60% → ไม่ลงขึ้น "หน้าเต็ม") · ต้องใช้ข้อมูลที่ยังไม่กรอก = ถามคำถามข้อเดียวก่อน1 ทั้งหน้า · 2 3 4 ชิ้นใหม่ชิ้นใหม่ (ชิ้นอื่นแค่ถูกดัน)
    
          
    ลากย้ายเปลี่ยนแค่ตำแหน่ง1ทั้งหน้า (เนื้อในไม่คิดใหม่)

    JSON มี 2 ไฟล์ ทำคนละหน้าที่
    
    
    ไฟล์การ์ดบอก "วางอะไร ตรงไหน ใส่อะไร" · ไฟล์แบบบอก "ชนิดนี้หน้าตาเป็นยังไง" · เจอกันด้วยชื่อไม่กี่ตัว
    
    
    
      
    
        
    การ์ด JSON1 ไฟล์ ต่อเจ้าของ
    
        
    • แอปเขียน (ทุกครั้งที่แก้) → saveStarCard
    • มีกรอบของทุกชิ้น + ข้อความ/รูปที่ใส่ + ข้อมูลเจ้าของ
    • ไม่มีหน้าตาของ widget เลย
    
      
    
      
    
        
    แบบ JSON (WidgetSpec)1 ไฟล์ ต่อชนิด
    
        
    • ดีไซเนอร์เขียน → widgetSpecs (82 ไฟล์)
    • เป็นต้นไม้ของกล่อง: แถว คอลัมน์ ข้อความ รูป ไอคอน
    • ไม่มีข้อมูลของใครเลย มีแต่ "ช่อง" ให้เติม
    
      
    
    
    
    
    
      
    ในการ์ด JSONในแบบ JSONได้อะไร
    
      
        
    items[].kind: "statPoster"→ไฟล์ statPoster.jsonเลือกแบบที่จะใช้วาด
    
        
    items[].w h→design: {w, h}ตัวคูณ s (ขั้น 2 ของตัววาด)
    
        
    —→"text": "MY Social"ป้ายตายตัวของดีไซน์ · widget ไม่มีข้อความที่ผู้ใช้พิมพ์ (ตัดแล้ว 19 ต.ค. 2569)
    
        
    owner.values.name→"slot": "name"ข้อมูลเจ้าของ (ชื่อ เบอร์ …) · แท็กไลน์สร้างจาก categories
    
        
    owner.intake.socials[]→"list": "socials"ช่องทางทั้งหมด → วาดซ้ำทีละช่อง
    
        
    photos["<id>#1"]→"photo": 1รูปในช่องที่ 1 ของชิ้น
    
        
    items[].surface: "glass"→"if": "papered"เลือกกิ่งของผังตามพื้นผิว

    มันทำงานยังไง: 4 เรื่องที่ต้องเข้าใจ
    
    
    ตัวเลขทุกตัวข้างล่างมาจากรอบที่ iOS วาด statPoster จริง
    

    
    
      
    1
    โหนดซ้ำ + if = เลือกกิ่ง
    ผังเขียนโหนดเดียวกันหลายแบบ แต่ละแบบมีเงื่อนไข ตัวคำนวณตัดโหนดที่เงื่อนไขเท็จทิ้ง ก่อนจัดกล่อง
    
      
    
        
    { "type": "text", "id": "line1", "if": "papered",  "color": "plateInk" }   ✓ วาด
{ "type": "text", "id": "line1", "if": "!papered", "color": "ink@0.95" }   ✗ ข้าม
    
        
    ชิ้นนี้พื้นผิว glass → papered จริง → ใช้หมึกครีมบนแผ่นเข้ม · ถ้าผู้ใช้เปลี่ยนเป็น clear จะได้อีกบรรทัด (หมึกของการ์ด) · ! = ไม่ · & = และ · | = หรือ
    
      
    
    
    

    
    
      
    2
    chips + list + each = วาดซ้ำตามข้อมูล
    each คือแม่พิมพ์ 1 ช่อง · ตัวคำนวณปั๊มแม่พิมพ์ 1 ครั้งต่อ 1 รายการใน list และแทน $field ด้วยค่าของรายการนั้น
    
      
    
        
    
          
    list "socials"{ name: "Instagram", followers: "24.8K" }{ name: "TikTok", followers: "58.3K" }{ name: "YouTube", followers: "8.9K" }
    
          
    × แม่พิมพ์ each
    "slot": "$followers"
    "slot": "$name"

    col + grow = แจกที่ว่างที่เหลือ
    คอลัมน์ body สูง 214 · เอาของที่ขนาดตายตัวลงก่อน ที่เหลือแจกให้ลูกที่มี grow ตามน้ำหนัก
    
      
    
        
    
          
    
          
    
            
    214 − ระยะขอบ (18 + 16) − ของตายตัว 164.67 = 15.32
    
            
    มี spacer "grow": 1 สองตัว → ได้ตัวละ 7.66
    
            
    ลากกรอบให้สูงขึ้น 40 → สองตัวนี้ได้เพิ่มตัวละ 20 · ของอื่นอยู่ที่เดิม · นี่คือเหตุผลที่ตอน resize ผังไม่เละ
    
          
    
        
    
      
    
    
    

    
    
      
    4
    ข้อความ: วัด → ย่อ → บอกจำนวนบรรทัด
    ตัวคำนวณวัดความกว้างตัวอักษรจากไฟล์ฟอนต์เดียวกันทุกเครื่อง แล้วตัดสินขนาด/บรรทัดเอง ห้ามให้ระบบย่อเอง
    
      
    
        
    { "text": "MY Social",
  "face": "noto", "weight": 800, "size": 22,
  "maxLines": 1, "shrinkTo": 12, "w": "66%" }
    
        
    
          
    1 หาข้อความป้ายตายตัว "MY Social" (ช่องข้อมูลอ่านจาก Model เช่น "slot": "name")
    
          
    2 หาความกว้าง66% ของ 330 = 217.8
    
          
    3 วัดnoto 800 ขนาด 22 → พอดี 1 บรรทัด ไม่ต้องย่อ
    
          
    4 ออกกล่อง 217.8 × 33.24 · size 22 · lines 1
    
        
    
        
    ถ้าข้อความยาวจนไม่พอ (เช่นชื่อยาว) → ลดทีละ 0.5 จนพอหรือถึง shrinkTo (12) · ยังไม่พอ → ตัววาดตัดท้ายด้วย … · ความสูง 33.24 = ความสูงบรรทัดของฟอนต์ noto ที่ 22
